    def test_cautious_fights_less_than_aggressive(self):

        headers = ['Fight', 'Evade']

            g = Game(seed=11)

            g.pending_event = {'code': 'e050', 'headers': headers, 'rows': []}

                                  for h in headers]}
            idx = bot.choose(prompt)
            self.assertEqual(headers[idx], expect,
                             f'{strategy} chose {headers[idx]}')
